STATS buy_qty  = SUM(order.filled_qty) WHERE order.side == "Buy",
      sell_qty = SUM(order.filled_qty) WHERE order.side == "Sell",
      total_filled_qty = SUM(order.filled_qty)
  BY order.account, order.symbol
| WHERE buy_qty > 0 AND sell_qty > 0
| EVAL net_pct = ABS(buy_qty - sell_qty)
                 / (buy_qty + sell_qty) * 100
| WHERE total_filled_qty > 27931289 AND net_pct < 0.3
