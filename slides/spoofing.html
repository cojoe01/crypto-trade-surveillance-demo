FROM logs-trades.orders-*
| WHERE order.status == "Canceled"
| STATS canceled_buys  = COUNT(*) WHERE order.side == "Buy",
        canceled_sells = COUNT(*) WHERE order.side == "Sell"
    BY order.account, order.symbol,
       time_bucket = BUCKET(order.timestamp, 60 second)
| EVAL total_cancellations = canceled_buys + canceled_sells,
       minority = LEAST(canceled_buys, canceled_sells),
       directional_ratio = minority / total_cancellations
| WHERE total_cancellations > 5
    AND directional_ratio < 0.15
